module mixed(input clk, output reg state, inout electrical p);
  real value;
  initial state = 0;
  always @(posedge clk) begin
    state <= ~state;
  end
  analog begin
    @(initial_step) value = 0.0;
    @(posedge clk) value = state ? 1.2 : 0.0;
    V(p) <+ transition(value, 0, 1n);
  end
endmodule

module bank(p, n);
  electrical [3:0] p;
  electrical n;
  genvar k;
  generate
    for (k = 0; k < 4; k = k + 1) begin : cells
      resistor #(.r(1k)) r0(.p(p[k]), .n(n));
    end
  endgenerate
  analog for (k = 0; k < 4; k = k + 1) I(p[k], n) <+ 1u;
endmodule

==================
Connectmodule and connectrules
==================

discipline logic;
  domain discrete;
enddiscipline
connectmodule e2l(input electrical a, output reg d);
  always @(cross(V(a) - 0.5)) d = V(a) > 0.5;
endmodule
connectrules bridge;
  connect e2l merged #(.threshold(0.5)) input electrical, output logic;
  connect electrical, logic resolveto electrical;
  connect logic resolveto exclude;
endconnectrules
