module capacitor(p, n);
  inout p, n;
  electrical p, n;
  parameter real c = 1p;
  analog begin
    I(p, n) <+ c * ddt(V(p, n));
    V(p) <+ idt(I(p, n), 0);
    V(n) <+ laplace_nd(V(p), {1.0}, {1.0, 1u});
    I(p) <+ white_noise(1e-18, "thermal") + flicker_noise(1e-20, 1.0, "flicker");
  end
endmodule

module piecewise(p);
  electrical p;
  integer mode;
  analog begin
    if (V(p) > 1.0)
      I(p) <+ 1m;
    else if (V(p) < -1.0)
      I(p) <+ -1m;
    else
      I(p) <+ 1m * V(p);
    case (mode)
      0, 1: V(p) <+ 0.0;
      2: begin I(p) <+ 1u; end
      default: ;
    endcase
  end
endmodule

module loops(p);
  electrical [3:0] p;
  integer i;
  real total;
  analog initial total = 0;
  analog begin : evaluate
    real tmp;
    tmp = 0;
    for (i = 0; i < 4; i = i + 1) begin
      tmp = tmp + V(p[i]);
      I(p[i]) <+ ddt(V(p[i]));
    end
    while (tmp < 1) tmp = tmp + 1;
    repeat (2) tmp = tmp / 2;
    $strobe("total = %g", tmp);
  end
endmodule

module functions(p);
  electrical p;
  analog function real limit;
    input x, bound;
    real x, bound, result;
    integer i;
    begin
      result = x;
      for (i = 0; i < 2; i = i + 1) result = result / 2;
      if (result > bound) limit = bound;
      else begin
        case (i)
          0: limit = 0;
          default: limit = result;
        endcase
      end
    end
  endfunction
  analog V(p) <+ limit(sin($abstime), 1.0);
endmodule

module events(p, clk);
  electrical p;
  input clk;
  real held;
  analog begin
    @(initial_step) held = 0.0;
    @(initial_step("dc", "tran") or final_step) $strobe("time=%g", $abstime);
    @(cross(V(p) - 0.5, +1, , 1u)) held = V(p);
    @(above(V(p) - 0.1) or timer(0, 1n)) held = V(p);
    @(posedge clk) held = 1.0;
    @(cross(V(p)), timer(1n)) ;
    V(p) <+ transition(held, 0, 1n, 1n);
  end
endmodule
